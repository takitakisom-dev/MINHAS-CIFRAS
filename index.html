function isChordToken(tok){ if(!tok) return false; let t = tok.replace(/^[\(\)\[\]\{\}]+|[\(\)\[\]\{\}\.,;:!]+$/g,'').trim(); if(t.length>24 || t.length<1) return false; if(!/^[A-G]/.test(t)) return false; // Aceita tudo que é cifra: Esus4/b9 Dm(b5/6/7/9/omit3) Am E7/#9 A#7/#11 Dm7 | Am (Am) return /^[A-G][#b]?[mM]?(?:sus|dim|aug|maj|min|add|omit|b5|#5|#9|b9|#11|b13|M7|7M)*[0-9]*(?:\([0-9A-Za-z#b\/\-]+\))*(\/[A-G]?[#b]?[0-9]*)?$/i.test(t.replace(/[\(\)]/g,'')) || /^(?:[A-G][#b]?m?(?:sus|dim|aug|maj7|m7|7M|7|9|6|11|13|add|b5|#5|#9|b9)*[0-9]*(\([A-Za-z0-9#b\/\-]+\))*(\/[A-G#\/b0-9#]+)?)$/.test(t) || /^[A-G][#b]?$/.test(t) || /^[\|]+$/.test(t); } function isCifraLine(line){ let l=line.trim(); if(!l) return false; if(/^\[.*\]$/.test(l)) return false; if(l.startsWith('2x') || l.startsWith('2X')) return true; // tipo 2x| E subo... // Se tem só acordes e pipes e barra let cleaned = l.replace(/\s+/g,' ').trim(); let tokens = cleaned.split(/\s+/).filter(Boolean); if(tokens.length===0) return false; // Se a linha inteira tem muito número e letra de cifra e pouca vogal let chordCount=0; for(let tk of tokens){ let c = tk.replace(/[\|]/g,''); if(!c) continue; if(isChordToken(c)) chordCount++; } // Linha com 1 ou 2 cifras tipo "Am" ou "Esus4/b9 Dm(b5/6/7/9/omit3)" tem que ser cifra if(tokens.length<=3 && chordCount>=1) return true; // Linha maior: 60% tem que ser cifra if(chordCount>=1 && (chordCount/tokens.length)>=0.5) return true; // Linha que começa com | é sempre cifra if(/^\|/.test(l)) return true; return false; }
